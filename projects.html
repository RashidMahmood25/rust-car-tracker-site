def monthly_installment(price, down_payment, annual_rate, years):
    loan = price - down_payment
    r = annual_rate / 100 / 12
    n = years * 12
    if r == 0:
        return loan / n
    return loan * r * (1 + r) ** n / ((1 + r) ** n - 1)

zakat-calculator-python

Calculate Zakat on cash, gold and savings.

Full source code — read it here, or download and run it in Python:

#!/usr/bin/env python3
"""Zakat calculator (simple, educational).

Asks for cash, bank balance, gold, silver and trade goods, subtracts
debts, then applies the standard 2.5% rate on wealth above the nisab.

Nisab note: this script uses the commonly quoted nisab values as
DEFAULTS only (gold: 87.48 grams, silver: 612.36 grams, valued at the
rates you enter). Gold/silver market rates change daily, so always
check the current local rate and enter it yourself. This tool does not
replace a scholar's ruling for complex cases (e.g. business stock,
agricultural produce, or debts of different kinds).
"""

GOLD_NISAB_GRAMS = 87.48
SILVER_NISAB_GRAMS = 612.36
ZAKAT_RATE = 0.025


def ask_float(prompt):
    while True:
        try:
            value = float(input(prompt).strip().replace(",", ""))
            if value < 0:
                print("  Please enter 0 or a positive number.")
                continue
            return value
        except ValueError:
            print("  Please enter a number.")


def calculate_zakat(cash, bank, gold_value, silver_value, trade_goods, debts):
    """Return (total_wealth, nisab, zakat_due)."""
    total_wealth = cash + bank + gold_value + silver_value + trade_goods - debts
    return total_wealth


def main():
    print("=== Zakat Calculator ===")
    print("All amounts in PKR. Enter 0 where not applicable.\n")

    cash = ask_float("Cash in hand (PKR): ")
    bank = ask_float("Bank balance (PKR): ")
    gold_grams = ask_float("Gold owned (grams, 0 if none): ")
    gold_rate = ask_float("Current gold rate per gram (PKR): ") if gold_grams else 0.0
    silver_grams = ask_float("Silver owned (grams, 0 if none): ")
    silver_rate = ask_float("Current silver rate per gram (PKR): ") if silver_grams else 0.0
    trade_goods = ask_float("Trade goods / stock value (PKR): ")
    debts = ask_float("Debts owed (PKR): ")

    gold_value = gold_grams * gold_rate
    silver_value = silver_grams * silver_rate
    net_wealth = calculate_zakat(cash, bank, gold_value, silver_value, trade_goods, debts)

    nisab_gold = GOLD_NISAB_GRAMS * gold_rate if gold_rate else 0.0
    nisab_silver = SILVER_NISAB_GRAMS * silver_rate if silver_rate else 0.0
    # Nisab is the LOWER of the gold and silver thresholds (the more
    # cautious, commonly used position); show both so the user can see.
    nisab = min(v for v in (nisab_gold, nisab_silver) if v > 0) if (nisab_gold or nisab_silver) else 0.0

    print("\n--- Result ---")
    print(f"Gold value:   PKR {gold_value:,.2f}")
    print(f"Silver value: PKR {silver_value:,.2f}")
    print(f"Net wealth (after debts): PKR {net_wealth:,.2f}")
    if nisab:
        print(f"Nisab (lower of gold/silver value): PKR {nisab:,.2f}")
    if net_wealth > 0 and nisab and net_wealth >= nisab:
        print(f"Zakat due (2.5%): PKR {net_wealth * ZAKAT_RATE:,.2f}")
    elif net_wealth <= 0:
        print("Zakat due: PKR 0 (net wealth is zero or negative).")
    else:
        print("Zakat due: PKR 0 (net wealth is below nisab).")
    print("\nNote: rates change daily — re-check gold/silver rates before paying.")


if __name__ == "__main__":
    main()

electricity-bill-calculator-pakistan-python

Estimate a Pakistan electricity bill — units, slabs and taxes.

Full source code — read it here, or download and run it in Python:

#!/usr/bin/env python3
"""Electricity bill estimator for Pakistan (residential, educational).

Uses a simplified slab structure the user can see and edit. Real
DISCO bills also include fixed charges, fuel price adjustment (FPA),
taxes/surcharges and slab changes that vary by distributor and month,
so this is an ESTIMATE of the energy charge only, not a replica bill.

Default slabs (PKR per unit, editable in SLABS below):
  0-100 units:   22.00
  101-200:       28.00
  201-300:       34.00
  301-400:       40.00
  401-500:       44.00
  501-600:       46.00
  601-700:       48.00
  701+:          50.00
These are PLACEHOLDER/illustrative rates, not the current notified
tariff. Check your DISCO's latest tariff before relying on the figure.
"""

# (upper_limit_inclusive, rate_per_unit). Last slab is open-ended.
SLABS = [
    (100, 22.00),
    (200, 28.00),
    (300, 34.00),
    (400, 40.00),
    (500, 44.00),
    (600, 46.00),
    (700, 48.00),
    (float("inf"), 50.00),
]


def energy_charge(units):
    """Slab-wise energy charge for `units` consumed."""
    remaining = units
    previous_limit = 0
    total = 0.0
    breakdown = []
    for limit, rate in SLABS:
        if remaining <= 0:
            break
        slab_units = min(remaining, limit - previous_limit)
        cost = slab_units * rate
        breakdown.append((previous_limit + 1, previous_limit + slab_units, rate, cost))
        total += cost
        remaining -= slab_units
        previous_limit = limit
    return total, breakdown


def main():
    print("=== Electricity Bill Estimator (Pakistan, residential) ===")
    print("Estimates the ENERGY CHARGE only — taxes, FPA and fixed")
    print("charges are NOT included. Rates below are illustrative.\n")
    while True:
        try:
            units = float(input("Units consumed (kWh): ").strip())
            if units < 0:
                print("  Please enter 0 or a positive number.")
                continue
            break
        except ValueError:
            print("  Please enter a number.")

    total, breakdown = energy_charge(units)
    print("\n--- Slab breakdown ---")
    for start, end, rate, cost in breakdown:
        end_label = int(end) if float(end).is_integer() else end
        label = f"{start}-{end_label}" if start != end else f"{start}"
        print(f"  {label} units @ PKR {rate:.2f} = PKR {cost:,.2f}")
    print(f"\nEstimated energy charge: PKR {total:,.2f}")
    print("Average per unit: PKR {:.2f}".format(total / units if units else 0))
    print("\nNote: add your DISCO's fixed charges, FPA, GST and other")
    print("surcharges for a bill closer to the real one.")


if __name__ == "__main__":
    main()

A simple command-line to-do list.

Full source code — read it here, or download and run it in Python:

#!/usr/bin/env python3
"""Simple command-line to-do list with file saving.

Tasks are stored in tasks.json next to this script, so your list
survives between runs. Each task has a title and a done/not-done flag.

Commands (typed at the prompt):
  add <task>   - add a new task
  list         - show all tasks
  done <n>     - mark task number n as done
  del <n>      - delete task number n
  quit         - save and exit
"""

import json
import os
import sys

TASKS_FILE = os.path.join(os.path.dirname(os.path.abspath(__file__)), "tasks.json")


def load_tasks():
    """Return the saved task list, or an empty list if none exists."""
    if not os.path.exists(TASKS_FILE):
        return []
    try:
        with open(TASKS_FILE, "r", encoding="utf-8") as f:
            data = json.load(f)
        if isinstance(data, list):
            return [t for t in data if isinstance(t, dict) and "title" in t]
    except (json.JSONDecodeError, OSError):
        print("Warning: saved tasks file was unreadable; starting fresh.")
    return []


def save_tasks(tasks):
    """Write the task list to tasks.json."""
    with open(TASKS_FILE, "w", encoding="utf-8") as f:
        json.dump(tasks, f, indent=2, ensure_ascii=False)


def show_tasks(tasks):
    if not tasks:
        print("No tasks yet. Use: add <task>")
        return
    for i, task in enumerate(tasks, 1):
        mark = "[x]" if task.get("done") else "[ ]"
        print(f"{i}. {mark} {task['title']}")


def main():
    tasks = load_tasks()
    print("=== To-Do List ===")
    print("Commands: add <task> | list | done <n> | del <n> | quit")
    show_tasks(tasks)
    while True:
        try:
            line = input("\n> ").strip()
        except (EOFError, KeyboardInterrupt):
            print()
            break
        if not line:
            continue
        parts = line.split(None, 1)
        cmd = parts[0].lower()
        arg = parts[1].strip() if len(parts) > 1 else ""
        if cmd == "quit":
            break
        elif cmd == "list":
            show_tasks(tasks)
        elif cmd == "add":
            if not arg:
                print("Usage: add <task>")
            else:
                tasks.append({"title": arg, "done": False})
                print(f"Added: {arg}")
        elif cmd in ("done", "del"):
            try:
                n = int(arg)
                if not 1 <= n <= len(tasks):
                    raise ValueError
            except ValueError:
                print(f"Usage: {cmd} <task number from list>")
                continue
            if cmd == "done":
                tasks[n - 1]["done"] = True
                print(f"Done: {tasks[n - 1]['title']}")
            else:
                removed = tasks.pop(n - 1)
                print(f"Deleted: {removed['title']}")
        else:
            print("Unknown command. Try: add, list, done, del, quit")
    save_tasks(tasks)
    print(f"Saved {len(tasks)} task(s) to tasks.json. Goodbye!")


if __name__ == "__main__":
    main()

Generate strong random passwords.

Full source code — read it here, or download and run it in Python:

#!/usr/bin/env python3
"""Password generator (Python).

Builds a random password of the length you choose, from the character
sets you pick. Uses the `secrets` module (made for security-sensitive
randomness) instead of the plain `random` module.

The generator guarantees at least one character from every set you
selected, so a password that asks for symbols will actually contain one.
"""

import secrets
import string


def ask_int(prompt, default, minimum):
    """Ask for a whole number; Enter accepts the default."""
    while True:
        raw = input(prompt).strip()
        if raw == "":
            return default
        try:
            value = int(raw)
            if value < minimum:
                print(f"  Please enter {minimum} or more.")
                continue
            return value
        except ValueError:
            print("  Please enter a whole number.")


def ask_yes_no(prompt, default=True):
    """Ask a y/n question; Enter accepts the default."""
    hint = "Y/n" if default else "y/N"
    while True:
        raw = input(f"{prompt} ({hint}): ").strip().lower()
        if raw == "":
            return default
        if raw in ("y", "yes"):
            return True
        if raw in ("n", "no"):
            return False
        print("  Please answer y or n.")


def generate_password(length, use_lower, use_upper, use_digits, use_symbols):
    """Return a password, or None if no character set was selected."""
    pools = []
    if use_lower:
        pools.append(string.ascii_lowercase)
    if use_upper:
        pools.append(string.ascii_uppercase)
    if use_digits:
        pools.append(string.digits)
    if use_symbols:
        pools.append("!@#$%^&*()-_=+")
    if not pools:
        return None
    if length < len(pools):
        return None
    # One guaranteed character from each selected set, then fill the rest.
    chars = [secrets.choice(pool) for pool in pools]
    all_chars = "".join(pools)
    chars += [secrets.choice(all_chars) for _ in range(length - len(chars))]
    # Shuffle so the guaranteed characters are not always at the front.
    for i in range(len(chars) - 1, 0, -1):
        j = secrets.randbelow(i + 1)
        chars[i], chars[j] = chars[j], chars[i]
    return "".join(chars)


def main():
    print("=== Password Generator ===")
    print("Press Enter to accept the default shown in [brackets].\n")
    length = ask_int("Password length [16]: ", 16, 4)
    use_lower = ask_yes_no("Include lowercase letters?", True)
    use_upper = ask_yes_no("Include UPPERCASE letters?", True)
    use_digits = ask_yes_no("Include digits?", True)
    use_symbols = ask_yes_no("Include symbols (!@#$...)?", True)

    password = generate_password(length, use_lower, use_upper, use_digits, use_symbols)
    if password is None:
        print("\nNo character set selected (or length too short) — nothing to generate.")
        print("Run again and pick at least one character set.")
        return
    print(f"\nYour password ({len(password)} characters):")
    print(password)
    print("\nTip: store it in a password manager — never reuse it across sites.")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""BMI Calculator — body mass index with WHO category.

Usage:
    python bmi_calculator.py
"""

CATEGORIES = [
    (18.5, "Underweight"),
    (25.0, "Normal weight"),
    (30.0, "Overweight"),
]


def calculate_bmi(weight_kg: float, height_m: float) -> float:
    """Return BMI = weight (kg) / height (m)^2."""
    if weight_kg <= 0 or height_m <= 0:
        raise ValueError("Weight and height must be positive numbers.")
    return weight_kg / (height_m ** 2)


def bmi_category(bmi: float) -> str:
    """Return the WHO category for a BMI value."""
    for limit, label in CATEGORIES:
        if bmi < limit:
            return label
    return "Obesity"


def main() -> None:
    print("=== BMI Calculator ===")
    try:
        weight = float(input("Weight in kg: "))
        height_cm = float(input("Height in cm: "))
        height_m = height_cm / 100
        bmi = calculate_bmi(weight, height_m)
    except ValueError as exc:
        print(f"Invalid input: {exc}")
        return
    print(f"\nYour BMI is: {bmi:.1f}")
    print(f"Category: {bmi_category(bmi)}")
    print("Note: BMI is a general guide only — it does not account for")
    print("muscle mass, age, or body composition. Consult a doctor for")
    print("personal health advice.")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""Currency Converter — convert between currencies using static rates.

Rates are STATIC demo rates (labelled in the README) — they are NOT live
market rates. For real transactions, always check a live rate source.

Usage:
    python currency_converter.py
"""

# Static demo rates: how many units of each currency equal 1 USD.
# Clearly labelled as demo data — update these values for real use.
RATES_TO_USD = {
    "USD": 1.0,
    "PKR": 278.0,
    "EUR": 0.92,
    "GBP": 0.79,
    "AED": 3.67,
    "SAR": 3.75,
    "INR": 83.5,
    "AUD": 1.52,
}


def convert(amount: float, from_currency: str, to_currency: str) -> float:
    """Convert amount from one currency to another via USD."""
    from_currency = from_currency.upper()
    to_currency = to_currency.upper()
    if from_currency not in RATES_TO_USD:
        raise ValueError(f"Unsupported currency: {from_currency}")
    if to_currency not in RATES_TO_USD:
        raise ValueError(f"Unsupported currency: {to_currency}")
    amount_in_usd = amount / RATES_TO_USD[from_currency]
    return amount_in_usd * RATES_TO_USD[to_currency]


def main() -> None:
    print("=== Currency Converter ===")
    print("Supported:", ", ".join(sorted(RATES_TO_USD)))
    print("NOTE: rates are static demo rates, not live market rates.\n")
    try:
        amount = float(input("Amount: "))
        from_cur = input("From currency (e.g. USD): ").strip()
        to_cur = input("To currency (e.g. PKR): ").strip()
        result = convert(amount, from_cur, to_cur)
    except ValueError as exc:
        print(f"Invalid input: {exc}")
        return
    print(f"\n{amount:.2f} {from_cur.upper()} = {result:.2f} {to_cur.upper()}")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""Car EMI vs Cash comparison calculator (Pakistan context).

Compares the total cost of buying a car on a bank loan (EMI) against
paying the full price in cash up front, so a buyer can see exactly how
much extra the loan costs in interest.

Run:  python3 emi_vs_cash.py
"""

MONTHS_PER_YEAR = 12


def monthly_emi(principal: float, annual_rate_percent: float, months: int) -> float:
    """Standard EMI formula. Handles the zero-interest edge case."""
    if months <= 0:
        raise ValueError("Loan tenure must be more than 0 months.")
    if principal < 0:
        raise ValueError("Loan amount cannot be negative.")
    monthly_rate = annual_rate_percent / 100.0 / MONTHS_PER_YEAR
    if monthly_rate == 0:
        return principal / months
    factor = (1 + monthly_rate) ** months
    return principal * monthly_rate * factor / (factor - 1)


def compare(car_price: float, down_payment: float,
            annual_rate_percent: float, years: int) -> dict:
    """Return a comparison of loan vs cash purchase for one car."""
    if car_price <= 0:
        raise ValueError("Car price must be positive.")
    if not 0 <= down_payment < car_price:
        raise ValueError("Down payment must be between 0 and the car price.")
    if years <= 0:
        raise ValueError("Loan tenure must be more than 0 years.")
    months = years * MONTHS_PER_YEAR
    loan_amount = car_price - down_payment
    emi = monthly_emi(loan_amount, annual_rate_percent, months)
    total_paid_on_loan = down_payment + emi * months
    total_interest = emi * months - loan_amount
    return {
        "car_price": car_price,
        "down_payment": down_payment,
        "loan_amount": loan_amount,
        "months": months,
        "monthly_emi": emi,
        "total_interest": total_interest,
        "total_cost_loan": total_paid_on_loan,
        "total_cost_cash": car_price,
        "extra_cost_of_loan": total_paid_on_loan - car_price,
    }


def fmt(amount: float) -> str:
    return f"Rs {amount:,.0f}"


def main() -> None:
    print("=== Car EMI vs Cash Comparison ===")
    try:
        car_price = float(input("Car price (Rs): "))
        down_payment = float(input("Down payment (Rs, 0 if none): "))
        annual_rate = float(input("Bank annual interest/markup rate (%): "))
        years = int(input("Loan tenure (years): "))
    except ValueError:
        print("Please enter valid numbers only.")
        return
    try:
        result = compare(car_price, down_payment, annual_rate, years)
    except ValueError as exc:
        print(f"Error: {exc}")
        return

    print("\n--- Loan option ---")
    print(f"Loan amount:        {fmt(result['loan_amount'])}")
    print(f"Monthly EMI:        {fmt(result['monthly_emi'])} for "
          f"{result['months']} months")
    print(f"Total interest:     {fmt(result['total_interest'])}")
    print(f"Total cost (loan):  {fmt(result['total_cost_loan'])}")
    print("\n--- Cash option ---")
    print(f"Total cost (cash):  {fmt(result['total_cost_cash'])}")
    print("\n--- Verdict ---")
    extra = result["extra_cost_of_loan"]
    if extra > 0:
        print(f"Loan costs {fmt(extra)} MORE than paying cash.")
    elif extra == 0:
        print("Loan and cash cost exactly the same (0% markup).")
    else:
        print(f"Loan costs {fmt(-extra)} LESS than cash (unusual — "
              "check the rate).")


if __name__ == "__main__":
    main()

Calculate an exact age from a date of birth.

Full source code — read it here, or download and run it in Python:

#!/usr/bin/env python3
"""Age calculator.

Given a date of birth, works out the exact age in years, months and days,
the total days lived, the weekday of birth, and the days until the next
birthday. Handles leap years correctly (including 29 February birthdays,
whose birthday falls on 28 February in non-leap years).

Run:  python3 age_calculator.py
"""

from datetime import date

DAYS_PER_YEAR = 365.25  # only used for display estimates, not exact age


def parse_date(text: str) -> date:
    """Parse a YYYY-MM-DD date, raising ValueError on bad input."""
    parts = text.strip().split("-")
    if len(parts) != 3:
        raise ValueError("Use the format YYYY-MM-DD, e.g. 1998-05-14.")
    year, month, day = (int(p) for p in parts)
    return date(year, month, day)


def exact_age(born: date, today: date) -> tuple[int, int, int]:
    """Return (years, months, days) of age on `today`.

    Borrows days/months like a calendar, so leap years are handled by
    the datetime library itself.
    """
    if born > today:
        raise ValueError("Date of birth cannot be in the future.")
    years = today.year - born.year
    months = today.month - born.month
    days = today.day - born.day
    if days < 0:
        months -= 1
        prev_month = today.month - 1 or 12
        prev_year = today.year - 1 if today.month == 1 else today.year
        days += _last_day_of_month(prev_year, prev_month)
    if months < 0:
        years -= 1
        months += 12
    return years, months, days


def _last_day_of_month(year: int, month: int) -> int:
    """Last day number of a given month (handles February leap years)."""
    if month == 12:
        return 31
    return (date(year, month + 1, 1) - date(year, month, 1)).days


def next_birthday(born: date, today: date) -> date:
    """Next birthday date on or after today.

    29 February birthdays are celebrated on 28 February in non-leap
    years.
    """
    try:
        candidate = date(today.year, born.month, born.day)
    except ValueError:  # 29 February in a non-leap year
        candidate = date(today.year, 2, 28)
    if candidate < today:
        try:
            candidate = date(today.year + 1, born.month, born.day)
        except ValueError:
            candidate = date(today.year + 1, 2, 28)
    return candidate

#!/usr/bin/env python3
"""Expense Tracker (Python) — a small command-line expense tracker.

Add expenses, list them, and see totals by category. Everything is saved
to expenses.csv in the same folder, so your data survives closing the
program. The CSV file can be opened in any spreadsheet app.

Run: python3 expense_tracker.py
"""

import csv
import os
from datetime import date

CSV_FILE = "expenses.csv"
FIELDNAMES = ["date", "category", "description", "amount_pkr"]

CATEGORIES = [
    "Food",
    "Transport",
    "Bills",
    "Shopping",
    "Health",
    "Entertainment",
    "Other",
]


def load_expenses():
    """Read expenses from expenses.csv; return a list of dicts."""
    if not os.path.exists(CSV_FILE):
        return []
    with open(CSV_FILE, newline="", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        expenses = []
        for row in reader:
            try:
                row["amount_pkr"] = float(row["amount_pkr"])
            except (KeyError, TypeError, ValueError):
                continue  # skip broken rows instead of crashing
            expenses.append(row)
        return expenses


def save_expenses(expenses):
    """Write all expenses back to expenses.csv."""
    with open(CSV_FILE, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=FIELDNAMES)
        writer.writeheader()
        writer.writerows(expenses)


def add_expense(expenses):
    """Ask for one expense and add it to the list."""
    print("\nCategories:")
    for i, cat in enumerate(CATEGORIES, start=1):
        print(f"  {i}. {cat}")
    choice = input("Category number (1-7): ").strip()
    try:
        category = CATEGORIES[int(choice) - 1]
    except (ValueError, IndexError):
        print("Invalid category number — expense not added.")
        return

    description = input("Description (e.g. Petrol for Alto): ").strip()
    if not description:
        print("Description cannot be empty — expense not added.")
        return

    amount_text = input("Amount in PKR: ").strip().replace(",", "")
    try:
        amount = float(amount_text)
    except ValueError:
        print("That is not a number — expense not added.")
        return
    if amount <= 0:
        print("Amount must be more than 0 — expense not added.")
        return

    expenses.append({
        "date": date.today().isoformat(),
        "category": category,
        "description": description,
        "amount_pkr": amount,
    })
    save_expenses(expenses)
    print(f"Added: {description} — Rs {amount:,.0f} ({category})")


def list_expenses(expenses):
    """Print all expenses, newest last, with a running total."""
    if not expenses:
        print("\nNo expenses yet. Use 'add' to record your first one.")
        return
    print()
    total = 0.0
    for i, e in enumerate(expenses, start=1):
        print(f"{i}. {e['date']} | {e['category']:<13} | "
              f"{e['description']:<30} | Rs {e['amount_pkr']:>10,.0f}")
        total += e["amount_pkr"]
    print("-" * 70)
    print(f"Total: Rs {total:,.0f} across {len(expenses)} expense(s)")


def show_summary(expenses):
    """Print total spent per category and the overall total."""
    if not expenses:
        print("\nNo expenses yet — nothing to summarise.")
        return
    totals = {}
    for e in expenses:
        totals[e["category"]] = totals.get(e["category"], 0.0) + e["amount_pkr"]
    grand_total = sum(totals.values())
    print("\nSpending by category:")
    for cat in sorted(totals, key=totals.get, reverse=True):
        share = totals[cat] / grand_total * 100
        print(f"  {cat:<13} Rs {totals[cat]:>10,.0f}  ({share:4.1f}%)")
    print(f"  {'TOTAL':<13} Rs {grand_total:>10,.0f}")


def delete_expense(expenses):
    """Delete one expense by its number."""
    if not expenses:
        print("\nNo expenses to delete.")
        return
    list_expenses(expenses)
    number = input("\nNumber to delete (or Enter to cancel): ").strip()
    if not number:
        return
    try:
        index = int(number) - 1
        removed = expenses.pop(index)
    except (ValueError, IndexError):
        print("Invalid number — nothing deleted.")
        return
    save_expenses(expenses)
    print(f"Deleted: {removed['description']} — Rs {removed['amount_pkr']:,.0f}")


def main():
    expenses = load_expenses()
    print("=== Expense Tracker ===")
    print(f"Loaded {len(expenses)} expense(s) from {CSV_FILE}.")
    print("Commands: add | list | summary | del | quit")

    while True:
        command = input("\n> ").strip().lower()
        if command == "add":
            add_expense(expenses)
        elif command == "list":
            list_expenses(expenses)
        elif command == "summary":
            show_summary(expenses)
        elif command == "del":
            delete_expense(expenses)
        elif command == "quit":
            save_expenses(expenses)
            print(f"Saved {len(expenses)} expense(s) to {CSV_FILE}. Goodbye!")
            break
        elif command in ("help", ""):
            print("Commands: add | list | summary | del | quit")
        else:
            print("Unknown command. Type 'help' to see the commands.")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""Web Scraping Demo (Python) — scrape book listings from a demo site.

Target site: https://books.toscrape.com/
This site exists ONLY for practising web scraping — it is a fake bookshop
with made-up books, built for learners. That is why it is safe to scrape:
no real website is hit, no real data is copied, and nothing is behind a
login.

What it does:
  1. Downloads the front page of books.toscrape.com
  2. Extracts each book's title, price (in £) and star rating
  3. Prints a tidy table in the terminal
  4. Saves the same data to books.csv (opens in any spreadsheet app)

Run: python3 scrape_books.py
Optional: python3 scrape_books.py 3   (scrape the first 3 catalogue pages)

Only the Python standard library is used — no pip install needed.
"""

import csv
import sys
import urllib.error
import urllib.request
from html.parser import HTMLParser

BASE_URL = "https://books.toscrape.com/"
CSV_FILE = "books.csv"

RATING_WORDS = {"One": 1, "Two": 2, "Three": 3, "Four": 4, "Five": 5}


class BookParser(HTMLParser):
    """Pulls title, price and rating out of each <article class='product_pod'>."""

    def __init__(self):
        super().__init__()
        self.books = []
        self._in_article = False
        self._in_price = False
        self._current = None

    def handle_starttag(self, tag, attrs):
        attrs = dict(attrs)
        if tag == "article" and attrs.get("class") == "product_pod":
            self._in_article = True
            self._current = {"title": "", "price": "", "rating": 0}
        if not self._in_article:
            return
        if tag == "img" and "alt" in attrs:
            self._current["title"] = attrs["alt"]
        if tag == "p" and attrs.get("class", "").startswith("star-rating"):
            word = attrs["class"].split()[-1]
            self._current["rating"] = RATING_WORDS.get(word, 0)
        if tag == "p" and attrs.get("class") == "price_color":
            self._in_price = True

    def handle_data(self, data):
        if self._in_article and self._in_price and self._current is not None:
            self._current["price"] = data.strip()

    def handle_endtag(self, tag):
        if tag == "p" and self._in_price:
            self._in_price = False
        if tag == "article" and self._in_article:
            if self._current and self._current["title"]:
                self.books.append(self._current)
            self._in_article = False
            self._current = None


def fetch_page(url):
    """Download one page; return its HTML text (or None on failure)."""
    request = urllib.request.Request(
        url, headers={"User-Agent": "web-scraping-demo-python/1.0 (learning project)"}
    )
    try:
        with urllib.request.urlopen(request, timeout=15) as response:
            return response.read().decode("utf-8")
    except urllib.error.URLError as e:
        print(f"Could not download {url}: {e}")
        return None


def scrape_pages(page_count):
    """Scrape the first `page_count` catalogue pages; return a list of books."""
    books = []
    for page in range(1, page_count + 1):
        if page == 1:
            url = BASE_URL
        else:
            url = f"{BASE_URL}catalogue/page-{page}.html"
        html = fetch_page(url)
        if html is None:
            break
        parser = BookParser()
        parser.feed(html)
        print(f"Page {page}: found {len(parser.books)} books")
        books.extend(parser.books)
    return books


def main():
    try:
        page_count = int(sys.argv[1]) if len(sys.argv) > 1 else 1
    except ValueError:
        print("Usage: python3 scrape_books.py [number_of_pages]")
        return
    if page_count < 1:
        print("Number of pages must be at least 1.")
        return

    print(f"Scraping {BASE_URL} ({page_count} page(s)) ...")
    books = scrape_pages(page_count)
    if not books:
        print("No books found — check your internet connection and try again.")
        return

    print()
    print(f"{'Title':<45} {'Price':>8}  {'Stars'}")
    print("-" * 62)
    for book in books:
        title = book["title"][:43]
        stars = "*" * book["rating"]
        print(f"{title:<45} {book['price']:>8}  {stars}")
    print("-" * 62)
    print(f"{len(books)} book(s) scraped.")

    with open(CSV_FILE, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=["title", "price_gbp", "rating_stars"])
        writer.writeheader()
        for book in books:
            writer.writerow({
                "title": book["title"],
                "price_gbp": book["price"].replace("£", ""),
                "rating_stars": book["rating"],
            })
    print(f"Saved to {CSV_FILE}.")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Blog Title Generator — catchy, SEO-friendly title ideas from one topic.

Usage:
    python blog_title_generator.py
    python blog_title_generator.py "electric cars in Pakistan"
    python blog_title_generator.py "electric cars in Pakistan" --count 5
    python blog_title_generator.py "electric cars in Pakistan" --category howto

How it works:
    You give it a topic (e.g. "electric cars in Pakistan").
    It fills that topic into proven title patterns (how-to, listicle,
    question, guide, mistake-avoidance, comparison ...) and prints a
    numbered list of ready-to-use titles. No internet needed.
"""

import argparse
import random
import sys

# ---------------------------------------------------------------------------
# Title patterns. {t} = the topic, {n} = a number.
# Each category is a list of (pattern, weight). Weight only affects the
# random shuffle order; every pattern can appear.
# ---------------------------------------------------------------------------
PATTERNS = {
    "howto": [
        "How to {t} — A Beginner's Guide",
        "How to {t} in {n} Simple Steps",
        "How to {t} Without Spending a Fortune",
        "How to Start With {t} (Even If You Are a Complete Beginner)",
        "How to {t}: The Complete 2026 Guide",
    ],
    "listicle": [
        "{n} Best Ways to {t}",
        "{n} Things You Must Know About {t}",
        "{n} Mistakes to Avoid When You {t}",
        "{n} Tips for {t} That Actually Work",
        "{n} Reasons Why {t} Matters in 2026",
    ],
    "question": [
        "Is {t} Worth It? An Honest Answer",
        "What Is {t} and Why Does Everyone Talk About It?",
        "Should You {t}? Pros and Cons Explained",
        "Why Is {t} So Popular Right Now?",
    ],
    "guide": [
        "The Ultimate Guide to {t}",
        "{t}: Everything You Need to Know",
        "A Complete Guide to {t} for Beginners",
        "{t} Explained in Simple Words",
    ],
    "comparison": [
        "{t} vs the Alternatives: Which One Is Right for You?",
        "{t}: Before and After — What Really Changes",
    ],
    "opinion": [
        "The Truth About {t} Nobody Tells You",
        "I Tried {t} for 30 Days — Here's What Happened",
        "Why Most People Get {t} Wrong",
    ],
}

CATEGORY_LABELS = {
    "howto": "How-To",
    "listicle": "Listicle",
    "question": "Question",
    "guide": "Guide",
    "comparison": "Comparison",
    "opinion": "Opinion / Story",
}


def clean_topic(topic):
    """Tidy the topic: strip whitespace, drop a trailing full stop,
    and capitalise the first letter for display."""
    topic = " ".join(topic.strip().split())
    if topic.endswith("."):
        topic = topic[:-1]
    return topic[:1].upper() + topic[1:] if topic else topic


def generate(topic, count=10, category=None, seed=None):
    """Return a list of (category, title) tuples."""
    topic = clean_topic(topic)
    if not topic:
        raise ValueError("Topic cannot be empty.")

    rng = random.Random(seed)
    categories = [category] if category else list(PATTERNS)
    pool = []
    for cat in categories:
        for pattern in PATTERNS[cat]:
            pool.append((cat, pattern))

    rng.shuffle(pool)
    titles = []
    used = set()
    for cat, pattern in pool:
        number = rng.choice([3, 5, 7, 10, 12, 15])
        title = pattern.format(t=topic, n=number)
        if title not in used:
            used.add(title)
            titles.append((cat, title))
        if len(titles) >= count:
            break
    return titles


def main(argv=None):
    parser = argparse.ArgumentParser(
        description="Generate catchy blog title ideas from a topic."
    )
    parser.add_argument("topic", nargs="?", help="blog topic, e.g. 'electric cars in Pakistan'")
    parser.add_argument("--count", type=int, default=10,
                        help="how many titles to generate (default 10, max 25)")
    parser.add_argument("--category", choices=sorted(PATTERNS),
                        help="only use one category: " + ", ".join(sorted(PATTERNS)))
    parser.add_argument("--seed", type=int, default=None,
                        help="fixed seed for repeatable output")
    args = parser.parse_args(argv)

    if args.topic is None:
        try:
            args.topic = input("Enter your blog topic: ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nNo topic given. Exiting.")
            return 1

    count = max(1, min(args.count, 25))
    try:
        titles = generate(args.topic, count=count,
                          category=args.category, seed=args.seed)
    except ValueError as exc:
        print(f"Error: {exc}")
        return 1

    print(f"\n{len(titles)} title ideas for: {clean_topic(args.topic)}\n")
    for i, (cat, title) in enumerate(titles, 1):
        print(f"{i:2}. [{CATEGORY_LABELS[cat]}] {title}")
    print("\nTip: pick one, then tweak the number or the year to fit your post.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""
Image Resizer — resize images from the command line (Pillow).

Usage:
    python image_resizer.py photo.jpg --width 800
    python image_resizer.py photo.jpg --height 600
    python image_resizer.py photo.jpg --width 800 --height 800 --keep-aspect
    python image_resizer.py photo.jpg --width 800 --output small.jpg
    python image_resizer.py photos/ --width 800            # whole folder
    python image_resizer.py photo.png --width 800 --quality 70 --format JPEG

What it does:
    - Resizes one image, or every image in a folder (jpg/jpeg/png/webp/bmp).
    - Keeps the aspect ratio by default (fits inside the given box).
    - Optional --keep-aspect off: stretches to the exact width x height.
    - Saves next to the original as <name>_resized.<ext> unless --output
      (or --out-dir for folders) is given.
    - --quality controls JPEG/WebP compression (1-95, default 85).

Install Pillow first:
    pip install Pillow
"""

import argparse
import sys
from pathlib import Path

from PIL import Image

SUPPORTED_EXTENSIONS = {".jpg", ".jpeg", ".png", ".webp", ".bmp"}


def target_size(img, width, height, keep_aspect):
    """Work out the (width, height) the image should become."""
    orig_w, orig_h = img.size
    if width and height:
        if not keep_aspect:
            return (width, height)
        scale = min(width / orig_w, height / orig_h)
        return (max(1, round(orig_w * scale)), max(1, round(orig_h * scale)))
    if width:
        scale = width / orig_w
        return (width, max(1, round(orig_h * scale)))
    if height:
        scale = height / orig_h
        return (max(1, round(orig_w * scale)), height)
    return img.size


def resize_one(src, width, height, keep_aspect, quality, out_format, output):
    """Resize one image file. Returns the output path."""
    with Image.open(src) as img:
        new_size = target_size(img, width, height, keep_aspect)
        resized = img.resize(new_size, Image.LANCZOS)

        fmt = (out_format or img.format or "JPEG").upper()
        if fmt == "JPG":
            fmt = "JPEG"
        ext = {"JPEG": ".jpg", "PNG": ".png", "WEBP": ".webp", "BMP": ".bmp"}.get(fmt, ".jpg")

        if output is None:
            output = src.with_stem(src.stem + "_resized").with_suffix(ext)
        else:
            output = Path(output)
            if output.suffix.lower() not in SUPPORTED_EXTENSIONS:
                output = output.with_suffix(ext)

        save_kwargs = {}
        if fmt in ("JPEG", "WEBP"):
            save_kwargs["quality"] = quality
            if resized.mode in ("RGBA", "LA", "P"):
                resized = resized.convert("RGB")

        resized.save(output, format=fmt, **save_kwargs)
        return output, img.size, new_size


def main(argv=None):
    parser = argparse.ArgumentParser(
        description="Resize images (single file or whole folder) with Pillow."
    )
    parser.add_argument("input", help="image file or folder of images")
    parser.add_argument("--width", type=int, help="target width in pixels")
    parser.add_argument("--height", type=int, help="target height in pixels")
    parser.add_argument("--keep-aspect", action="store_true", default=True,
                        help="fit inside width x height without stretching (default)")
    parser.add_argument("--stretch", dest="keep_aspect", action="store_false",
                        help="stretch to the exact width x height")
    parser.add_argument("--quality", type=int, default=85,
                        help="JPEG/WebP quality 1-95 (default 85)")
    parser.add_argument("--format", dest="out_format",
                        choices=["JPEG", "PNG", "WEBP", "BMP"],
                        help="output format (default: keep the original)")
    parser.add_argument("--output", help="output file (single image only)")
    parser.add_argument("--out-dir", help="output folder (folder mode only)")
    args = parser.parse_args(argv)

    if not args.width and not args.height:
        print("Error: give --width and/or --height.")
        return 1
    if args.width is not None and args.width <= 0 or args.height is not None and args.height <= 0:
        print("Error: width and height must be positive numbers.")
        return 1
    if not 1 <= args.quality <= 95:
        print("Error: --quality must be between 1 and 95.")
        return 1

    src = Path(args.input)
    if not src.exists():
        print(f"Error: {src} does not exist.")
        return 1

    if src.is_dir():
        if args.output:
            print("Error: --output works for a single image only; use --out-dir for folders.")
            return 1
        files = sorted(p for p in src.iterdir()
                       if p.is_file() and p.suffix.lower() in SUPPORTED_EXTENSIONS)
        if not files:
            print(f"No supported images found in {src} (jpg, png, webp, bmp).")
            return 1
        out_dir = Path(args.out_dir) if args.out_dir else src
        out_dir.mkdir(parents=True, exist_ok=True)
        done, failed = 0, 0
        for f in files:
            try:
                out, old, new = resize_one(
                    f, args.width, args.height, args.keep_aspect,
                    args.quality, args.out_format,
                    out_dir / f.name if args.out_dir else None)
                print(f"{f.name}: {old[0]}x{old[1]} -> {new[0]}x{new[1]}  ({out})")
                done += 1
            except Exception as exc:  # noqa: BLE001 - report per file, keep going
                print(f"{f.name}: FAILED ({exc})")
                failed += 1
        print(f"\nDone: {done} resized, {failed} failed.")
        return 0 if failed == 0 else 1

    # Single file
    try:
        out, old, new = resize_one(src, args.width, args.height,
                                   args.keep_aspect, args.quality,
                                   args.out_format, args.output)
    except FileNotFoundError:
        print(f"Error: {src} is not a readable image file.")
        return 1
    except Exception as exc:  # noqa: BLE001 - Pillow raises many error types
        print(f"Error: could not resize {src} ({exc})")
        return 1
    print(f"{src.name}: {old[0]}x{old[1]} -> {new[0]}x{new[1]}")
    print(f"Saved: {out}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
